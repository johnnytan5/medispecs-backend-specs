================================================================================
CAREGIVER FALL DETECTION POLLING GUIDE
For Frontend/Mobile App Integration
================================================================================

OVERVIEW:
---------
Poll the emergency status endpoint every 5 seconds to monitor for falls in
real-time. The system uses a 3-state response model to track fall events
and user confirmation status.

================================================================================
ENDPOINT
================================================================================

URL: GET /accelerometer/emergency/status

Via ngrok: https://YOUR-NGROK-URL.ngrok.io/accelerometer/emergency/status
Via local: http://localhost:8000/accelerometer/emergency/status

Poll Interval: Every 5 seconds

================================================================================
RESPONSE STRUCTURE
================================================================================

{
  "monitoring": true,              // Is fall detection active?
  "current_state": "idle",         // idle | free_fall | impact | inactivity
  "latest_fall": {                 // null if no fall, or fall object:
    "timestamp": "2025-11-11T20:15:30.123456Z",  // When fall was detected
    "freefall_g": 0.35,            // Free fall magnitude (G)
    "impact_g": 2.8,               // Impact magnitude (G)
    "inactivity_sec": 5.2,         // Inactivity duration (seconds)
    "user_response": null,         // null | "CONFIRMED" | "NO_RESPONSE"
    "response_text": "okay",       // What user actually said (or null)
    "acknowledged": false,         // Has system finished handling fall?
    "simulated": false             // true if test fall, false if real
  },
  "current_readings": {            // Real-time sensor data
    "acceleration": {
      "x": 0.02, "y": -0.01, "z": 0.98, "total": 0.98
    },
    "state": "idle",
    "time_in_state": 42.1
  }
}

================================================================================
THE 3-STATE MODEL
================================================================================

user_response can be one of THREE values:

1. null
   ─────
   MEANING: Fall just detected, waiting for user to say "okay"
   TIMING:  From 0-15 seconds after fall
   ACTION:  Show warning to caregiver ("Fall detected, waiting...")
   
2. "CONFIRMED"
   ────────────
   MEANING: User said "okay" within 15 seconds
   TIMING:  Anytime from 0-15 seconds after fall
   ACTION:  Clear warning, log event as informational
   
3. "NO_RESPONSE"
   ──────────────
   MEANING: 15 seconds passed, user did NOT say "okay"
   TIMING:  After 15 second timeout
   ACTION:  🚨 SEND EMERGENCY ALERT! Call 911, notify caregiver!

================================================================================
POLLING LOGIC (PSEUDOCODE)
================================================================================

// Poll every 5 seconds
setInterval(async () => {
  try {
    const response = await fetch('https://YOUR-NGROK-URL/accelerometer/emergency/status');
    const data = await response.json();
    
    // Check if fall detection is active
    if (!data.monitoring) {
      showStatus("Fall detection is offline");
      return;
    }
    
    // No fall detected
    if (!data.latest_fall) {
      showStatus("No falls detected - all clear");
      return;
    }
    
    // Fall detected - handle based on user_response state
    const fall = data.latest_fall;
    
    if (fall.user_response === null) {
      // STATE 1: Waiting for user confirmation (0-30s)
      showWarning({
        type: "waiting",
        message: `Fall detected at ${formatTime(fall.timestamp)}`,
        subtitle: "Waiting for user response...",
        icon: "⏳"
      });
    }
    else if (fall.user_response === "CONFIRMED") {
      // STATE 2: User confirmed they're okay
      showSuccess({
        type: "confirmed",
        message: `User confirmed: "${fall.response_text}"`,
        subtitle: `Fall at ${formatTime(fall.timestamp)} - User is okay`,
        icon: "✅",
        dismissable: true
      });
    }
    else if (fall.user_response === "NO_RESPONSE") {
      // STATE 3: EMERGENCY! User did not respond
      sendEmergencyAlert({
        type: "emergency",
        message: "🚨 FALL EMERGENCY - NO RESPONSE",
        subtitle: `Fall detected at ${formatTime(fall.timestamp)}`,
        fallData: fall,
        actions: ["Call 911", "Call Family", "View Video"]
      });
    }
    
  } catch (error) {
    console.error("Polling error:", error);
    showStatus("Connection lost");
  }
}, 5000); // Poll every 5 seconds

================================================================================
TIMING DIAGRAM
================================================================================

t=0s   : Fall detected
         └─> latest_fall.user_response = null
         └─> TTS: "Fall detected! Are you okay? Please say okay"
         └─> System listens for 15 seconds...

t=5s   : Caregiver polls endpoint
         └─> Response: user_response = null
         └─> Frontend shows: "⏳ Waiting for response..."

t=10s  : User says "okay"
         └─> latest_fall.user_response = "CONFIRMED"
         └─> latest_fall.response_text = "okay"
         └─> TTS: "Glad you're okay!"

t=15s  : Caregiver polls endpoint
         └─> Response: user_response = "CONFIRMED"
         └─> Frontend shows: "✅ User is okay"

--- OR (if user doesn't respond) ---

t=15s  : Timeout!
         └─> latest_fall.user_response = "NO_RESPONSE"

t=20s  : Caregiver polls endpoint
         └─> Response: user_response = "NO_RESPONSE"
         └─> Frontend: 🚨 EMERGENCY ALERT!

================================================================================
IMPLEMENTATION TIPS
================================================================================

1. HANDLE NETWORK FAILURES
   -------------------------
   - Retry failed polls immediately (max 3 retries)
   - Show "Connection lost" if polling fails
   - Alert caregiver if connection is lost for >15 seconds

2. DEBOUNCE ALERTS
   ----------------
   - Only send ONE emergency alert per fall event
   - Track fall by timestamp to avoid duplicate alerts
   - Example: Store last alerted timestamp in localStorage

3. SHOW COUNTDOWN TIMER (Optional)
   --------------------------------
   - Calculate elapsed time: now - fall.timestamp
   - Show countdown: "Waiting... (10s remaining)"
   - More urgent as it approaches 15s

4. VIDEO INTEGRATION
   -------------------
   - Fall-triggered videos have "FALL_" prefix
   - After fall, fetch video: GET /timelapse/videos?userId=u_123
   - Filter for videos starting with "FALL_"
   - Allow caregiver to view fall footage

5. PERSISTENT NOTIFICATION
   -------------------------
   - Keep warning visible until fall is resolved
   - Don't auto-dismiss during null state
   - Only dismiss after CONFIRMED or caregiver acknowledges

6. ACCESSIBILITY
   --------------
   - Use high-contrast colors for emergency state (red)
   - Large, clear action buttons ("Call 911", "Dismiss")
   - Sound alert on NO_RESPONSE (if app is in foreground)

REACT EXAMPLE (TYPESCRIPT)
================================================================================

type UserResponse = null | "CONFIRMED" | "NO_RESPONSE";

interface FallEvent {
  timestamp: string;
  freefall_g: number;
  impact_g: number;
  inactivity_sec: number;
  user_response: UserResponse;
  response_text: string | null;
  acknowledged: boolean;
}

interface EmergencyStatus {
  monitoring: boolean;
  current_state: string;
  latest_fall: FallEvent | null;
}

// React Hook
function useFallDetection(ngrokUrl: string) {
  const [status, setStatus] = useState<EmergencyStatus | null>(null);
  const [alertedFalls, setAlertedFalls] = useState<Set<string>>(new Set());

  useEffect(() => {
    const interval = setInterval(async () => {
      try {
        const res = await fetch(`${ngrokUrl}/accelerometer/emergency/status`);
        const data: EmergencyStatus = await res.json();
        setStatus(data);

        // Handle emergency
        if (data.latest_fall?.user_response === "NO_RESPONSE") {
          const fallId = data.latest_fall.timestamp;
          
          // Only alert once per fall
          if (!alertedFalls.has(fallId)) {
            sendEmergencyAlert(data.latest_fall);
            setAlertedFalls(prev => new Set(prev).add(fallId));
          }
        }
      } catch (error) {
        console.error("Polling failed:", error);
      }
    }, 5000);

    return () => clearInterval(interval);
  }, [ngrokUrl, alertedFalls]);

  return status;
}

// Usage
function CaregiverDashboard() {
  const status = useFallDetection("https://your-ngrok-url.ngrok.io");

  if (!status?.monitoring) {
    return <Alert type="warning">Fall detection offline</Alert>;
  }

  if (!status.latest_fall) {
    return <Status>No falls detected - all clear ✅</Status>;
  }

  const fall = status.latest_fall;

  if (fall.user_response === null) {
    return (
      <Alert type="warning" icon="⏳">
        <h3>Fall Detected - Waiting for Response</h3>
        <p>Time: {formatTime(fall.timestamp)}</p>
        <p>Impact: {fall.impact_g}G</p>
        <Countdown start={fall.timestamp} duration={15} />
      </Alert>
    );
  }

  if (fall.user_response === "CONFIRMED") {
    return (
      <Alert type="success" icon="✅" dismissable>
        <h3>User Confirmed: "{fall.response_text}"</h3>
        <p>Fall at {formatTime(fall.timestamp)} - User is okay</p>
      </Alert>
    );
  }

  if (fall.user_response === "NO_RESPONSE") {
    return (
      <Alert type="emergency" icon="🚨" persistent>
        <h1>FALL EMERGENCY - NO RESPONSE</h1>
        <p>Fall detected at {formatTime(fall.timestamp)}</p>
        <p>Impact: {fall.impact_g}G | Inactivity: {fall.inactivity_sec}s</p>
        <ButtonGroup>
          <Button variant="danger" onClick={call911}>Call 911</Button>
          <Button variant="warning" onClick={callFamily}>Call Family</Button>
          <Button variant="secondary" onClick={viewVideo}>View Video</Button>
        </ButtonGroup>
      </Alert>
    );
  }
}
